private function applySubsidy3PlanSplit($ovum, $embryo, $implant, $applyPlan, $insideOperation)
{
if (strpos($applyPlan, 'C.') !== false) {
$ovum = 0; $embryo = 0;
} else if (strpos($applyPlan, 'D.') !== false || strpos($applyPlan, 'E.') !== false
|| strpos($applyPlan, 'F.') !== false) {
$implant = 0; $embryo = 0;
} else if (strpos($applyPlan, '1.') !== false) {
if ($insideOperation === '取卵') { $implant = 0; $embryo = 0; }
else if ($insideOperation === '植入') { $ovum = 0; $embryo = 0; }
else { $ovum = 0; $embryo = 0; $implant = 0; }
}
// 形成併入取卵；只取卵／只植入時上面已把形成清 0
+ $ovum = intval($ovum) + intval($embryo);
+ $embryo = 0;
+ $implant = intval($implant);
return array("ovum"=>$ovum,"embryo"=>$embryo,"implant"=>$implant,"total"=>$ovum+$implant);
}
